### 流水线推进函数

```go
// advancePipeline 是流水线的状态机驱动函数
func (s *ProjectService) advancePipeline(projectID uint) {
    var p models.Project
    s.db.First(&p, projectID)
    
    switch p.PipelineStage {
    case "plan":
        // 阶段 1: 生成创作方案
        fresh, _ := s.GeneratePlan(&p)
        s.GenerateAllPortraits(fresh)      // 触发角色标准像生成
        s.GenerateAllAssetImages(fresh, "") // 触发资产参考图生成
        s.db.Model(&p).Update("pipeline_stage", "script")
        go s.advancePipeline(fresh.ID)      // 递归推进
        
    case "script":
        // 阶段 2: 生成剧本
        s.GenerateScript(&p, episodeN)
        s.db.Model(&p).Update("pipeline_stage", "images")
        go s.advancePipeline(p.ID)
        
    case "images":
        // 阶段 3: 生成画面
        // 等待资产参考图就绪
        if s.pendingSceneAssetRefs(p.ID, episodeN, p.Generation) > 0 {
            s.GenerateAllAssetImages(&p, "")
            return  // 等待下一轮
        }
        s.GenerateAllImages(&p, episodeN)  // 批量生成
        // 继续等待...
        
    case "videos":
        // 阶段 4: 生成视频
        s.GenerateAllVideos(&p, episodeN)
        s.GenerateProjectDubs(&p)  // 异步配音
        s.db.Model(&p).Update("pipeline_stage", "merge")
        go s.advancePipeline(p.ID)
        
    case "merge":
        // 阶段 5: 合并成片
        s.CreateMergeTask(&p, sceneIDs, true, true)
    }
}
```

### 创作方案生成流程

```go
func (s *ProjectService) GeneratePlan(p *models.Project) (*models.Project, error) {
    // 1. 构建 Prompt
    system := planSystemPrompt()  // 专业短剧方法论
    user := buildPlanUserPrompt(p)
    
    // 2. 调用火山引擎文生文
    raw, err := s.volc.Chat(system, user)
    
    // 3. 解析 JSON
    plan, err := parsePlanJSON(raw)
    
    // 4. 事务落库
    s.db.Transaction(func(tx *gorm.DB) error {
        planJSON, _ := json.Marshal(plan)
        tx.Model(p).Updates(map[string]any{
            "plan":  string(planJSON),
            "title": plan.Title,  // 覆盖剧名
            "status": "plan_done",
        })
        
        // 5. 抽取角色为资产
        s.upsertCharactersFromPlan(p, plan)
        return nil
    })
    
    // 6. 触发角色标准像 + 资产参考图生成
    s.GenerateAllPortraits(p)
    s.GenerateAllAssetImages(p, "")
    
    return p, nil
}
```

### 场景落库

```go
func (s *ProjectService) generateScriptCore(p *models.Project, episodeN int, raw string) error {
    res, _ := parseScriptJSON(raw)
    
    // 事务
    s.db.Transaction(func(tx *gorm.DB) error {
        // 1. 删除旧场景
        tx.Where("project_id = ? AND episode_n = ?", p.ID, episodeN).Delete(&models.Scene{})
        
        // 2. 创建新场景
        for i, sc := range res.Scenes {
            scene := models.Scene{
                ProjectID: p.ID,
                EpisodeN: episodeN,
                Order: i + 1,
                Generation: newGeneration,
                Title: sc.Title,
                Content: sc.Content,
                ImagePrompt: sc.ImagePrompt,
                Duration: normalizeSceneDuration(sc.Duration),
                Characters: joinSceneCharacters(sc.Characters),
                LocationName: sc.Location,
                Props: joinSceneCharacters(sc.Props),
                Status: "pending",
            }
            tx.Create(&scene)
            
            // 3. 创建对白
            for j, d := range sc.Dialogues {
                tx.Create(&models.Dialogue{
                    SceneID: scene.ID,
                    Character: d.Character,
                    Text: d.Text,
                    Order: j + 1,
                })
            }
        }
        
        // 4. 更新剧本
        scripts := parseScripts(p.Scripts)
        scripts[episodeN] = res.Script
        tx.Model(p).Updates(map[string]any{
            "scripts": marshalJSON(scripts),
            "status": "script_done",
            "pipeline_stage": "images",
        })
        return nil
    })
    
    // 5. 自动补建资产（场景引用的道具/地点未建卡时）
    s.upsertAssetsFromScenes(p.ID, res.Scenes)
}
```

### Prompt 拼装策略

```go
func (s *ProjectService) buildSceneImagePrompt(sc *models.Scene) string {
    parts := make([]string, 0, 6)
    
    // 1. 画风约束
    if desc := styleDescriptor(p.Style); desc != "" {
        parts = append(parts, desc)
    }
    
    // 2. 视觉基准
    if vb := strings.TrimSpace(p.VisualBible); vb != "" {
        parts = append(parts, "全片视觉基准：" + vb)
    }
    
    // 3. 角色设定（跨场景一致性）
    if charCtx := s.characterContextForScene(sc); charCtx != "" {
        parts = append(parts, charCtx)
    }
    
    // 4. 道具/场景设定
    if assetCtx := s.assetContextForScene(sc); assetCtx != "" {
        parts = append(parts, assetCtx)
    }
    
    // 5. 当前分镜提示词
    parts = append(parts, "当前分镜：" + sc.ImagePrompt)
    
    return strings.Join(parts, "\n")
}
```

### 角色一致性注入

```go
func (s *ProjectService) characterContextForScene(sc *models.Scene) string {
    names := parseSceneCharacters(sc.Characters)
    var chars []models.Character
    s.db.Where("project_id = ? AND name IN ? AND portrait != ''", sc.ProjectID, names).Find(&chars)
    
    lines := make([]string, 0, len(names))
    for _, n := range names {
        c := chars[n]
        desc := c.Trait + "；" + c.Style
        lines = append(lines, "- " + c.Name + "：" + desc)
    }
    
    return "【角色设定（必须严格遵守，保证人物一致）】\n" + strings.Join(lines, "\n")
}
```

### 参考图注入

```go
func (s *ProjectService) generateClaimedSceneImage(sc *models.Scene, token string) error {
    prompt := s.buildSceneImagePrompt(sc)
    
    // 收集角色标准像 + 道具/场景参考图
    refs, names, _ := s.scenePortraitRefs(sc)      // 角色标准像
    assetRefs, assetLines := s.sceneAssetRefs(sc) // 道具/场景参考图
    refs = append(refs, assetRefs...)
    names = append(names, assetLines...)
    
    if len(refs) > 0 {
        prompt += "\n参考图说明：\n" + strings.Join(names) +
            "\n请严格保持图1至图N中各参考主体的外观完全一致。"
    }
    
    // 调用火山文生图（带参考图）
    data, err := s.volc.GenerateImageRefs(prompt, size, refs)
    
    // 保存文件
    path, _, _ := s.upload.SaveFile(taskID, "image", name, data)
    
    // 更新场景（验证 token 防过期）

func (s *ProjectService) runMerge(p *models.Project, mt *models.MergeTask, keepAudio, concatFirst bool) error {
    // 1. 获取场景视频列表
    scenes := getScenesByIDs(mt.SceneIDs)
    
    // 2. 构造 concat 文件列表
    var concatLines []string
    for _, sc := range scenes {
        videoPath := filepath.Join(
            s.cfg.Comfy.ComfyDir,
            "output_workers",
            fmt.Sprintf("gpu%d", *sc.VideoGPU),
            sc.VideoFile,
        )
        // 检测音轨（检查是否存在音频流）
        hasAudio := checkAudioTrack(videoPath)
        if hasAudio && keepAudio {
            concatLines = append(concatLines, fmt.Sprintf("file '%s'", videoPath))
        } else if !hasAudio {
            // 无音轨的视频用 anullsrc 填充
            concatLines = append(concatLines, fmt.Sprintf(
                "file '%s'\nfile '%s'",
                videoPath, generateSilentAudio(sc.Duration),
            ))
        }
    }
    
    // 3. 写入临时文件列表
    concatFile := "/tmp/merge_list_" + mt.TaskID + ".txt"
    writeConcatList(concatFile, concatLines)
    
    // 4. 执行 ffmpeg 合并
    outputFile := filepath.Join(
        s.cfg.Comfy.ComfyDir,
        "output_workers/gpu0",
        mt.OutputFile,
    )
    
    cmd := fmt.Sprintf(`
        ffmpeg -y -f concat -safe 0 -i %s \
        -c:v libx264 -pix_fmt yuv420p -movflags +faststart \
        -c:a aac -b:a 192k \
        %s
    `, concatFile, outputFile)
    
    s.remote.Run(cmd)
    
    // 5. 更新状态
    s.db.Model(mt).Updates(map[string]any{
        "status": "success",
    })
}
```

// 分镜场景引用的道具/地点未建卡时自动补建
func (s *ProjectService) upsertAssetsFromScenes(projectID uint, scenes []scriptScene) {
    // 收集所有 location 和 props
    locations := make(map[string]bool)
    props := make(map[string]bool)
    
    for _, sc := range scenes {
        locations[sc.Location] = true
        for _, p := range sc.Props {
            props[p] = true
        }
    }
    
    // 查询已有资产
    var existing []models.Asset
    s.db.Where("project_id = ? AND kind IN ?", projectID, []string{"location", "prop"}).Find(&existing)
    existingNames := make(map[string]bool)
    for _, a := range existing {
        existingNames[a.Kind + ":" + a.Name] = true
    }
    
    // 补建缺失资产
    for loc := range locations {
        if loc == "" || existingNames["location:"+loc] {
            continue
        }
        s.db.Create(&models.Asset{
            ProjectID: projectID,
            Kind: "location",
            Name: loc,
            Source: "auto",
        })
    }
    
    for prop := range props {
        if prop == "" || existingNames["prop:"+prop] {
            continue
        }
        s.db.Create(&models.Asset{
            ProjectID: projectID,
            Kind: "prop",
            Name: prop,
            Source: "auto",
        })
    }
}

func (s *ProjectService) WatchSceneVideos() {
    ticker := time.NewTicker(3 * time.Second)  // 每 3 秒轮询
    for {
        select {
        case <-s.stopped:
            return
        case <-ticker.C:
            s.syncSceneVideos()  // 同步视频任务状态
            s.advanceAutoPipelines()  // 推进自动流水线
        }
    }
}

func (s *ProjectService) syncSceneVideos() {
    var scenes []models.Scene
    s.db.Where("status IN ?", []string{"video_pending", "video_running"}).Find(&scenes)
    
    for _, sc := range scenes {
        var task models.Task
        s.db.Where("task_id = ?", sc.VideoTaskID).First(&task)
        
        switch task.Status {
        case "running", "queued":
            // 仍在执行
        case "success":
            // 成功：回写视频文件
            file, gpu := resultVideoOf(&task)
            s.db.Model(&sc).Updates(map[string]any{
                "status": "video_ready",
                "video_file": file,
                "video_gpu": gpu,
            })
        case "failed", "cancelled":
            // 失败：重试或标记
            s.retryOrFailVideo(&sc, task.Error)
        }
    }
}

    // 合并任务 → 重置为 pending
    s.db.Model(&MergeTask{}).Where("status = ?", "running").
        Update("status", "pending")
    
    // 流水线阶段 → 回退
    s.db.Model(&Project{}).Where("pipeline_stage = ?", "script_running").
        Update("pipeline_stage", "script")
}

func (s *ProjectService) claimSceneImage(sc *models.Scene) (string, error) {
    token := fmt.Sprintf("%d-%d", sc.ID, time.Now().UnixNano())
    
    // 原子更新：只有符合条件才更新
    claim := s.db.Model(&Scene{}).

    }
    return token, nil
}

## 总结

漫剧流水线是一个完整的多阶段自动化生产系统，核心设计要点：

| 特性 | 实现方式 |
|------|---------|
| **流水线状态机** | `pipeline_stage` 字段驱动，`advancePipeline()` 递归推进 |
| **幂等性** | 令牌机制 + 原子更新，防止重复提交和过期回写 |
| **一致性** | 三层保障：文本描述 + 视觉参考 + 模型能力 |
| **并发控制** | 信号量限流（文生图 3 并发）+ 去重标记 |
| **故障恢复** | 后台轮询 + 状态回退 + 自动重试 |
| **资产管理** | 自动补建 + 版本隔离（generation 字段） |
